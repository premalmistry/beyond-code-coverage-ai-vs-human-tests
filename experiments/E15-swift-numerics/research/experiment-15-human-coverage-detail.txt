   48|     16|  internal static func rescaledDivide(_ z: Complex, _ w: Complex) -> Complex {
   49|     16|    if w.isZero { return .infinity }
   50|     12|    if !w.isFinite { return .zero }
   51|     12|    //  Scaling algorithm adapted from Doug Priest's "Efficient Scaling for
   52|     12|    //  Complex Division":
   53|     12|    if w.magnitude < .leastNormalMagnitude {
   54|      4|      //  A difference from Priest's algorithm is that he didn't have to worry
   55|      4|      //  about types like Float16, where the significand width is comparable
   56|      4|      //  to the exponent range, such that |leastNormalMagnitude|^(-¾) isn't
   57|      4|      //  representable (e.g. for Float16 it would want to be 2¹⁸, but the
   58|      4|      //  largest allowed exponent is 15). Note that it's critical to use zʹ/wʹ
   59|      4|      //  after rescaling to avoid this, rather than falling through into the
   60|      4|      //  normal rescaling, because otherwise we might end up back in the
   61|      4|      //  situation where |w| ~ 1.
   62|      4|      let s = 1/(RealType(RealType.radix) * .leastNormalMagnitude)
   63|      4|      let wʹ = w.multiplied(by: s)
   64|      4|      let zʹ = z.multiplied(by: s)
   65|      4|      return zʹ / wʹ
   66|      8|    }
   67|      8|    //  Having handled that case, we proceed pretty similarly to Priest:
   68|      8|    //
   69|      8|    //  1. Choose real scale s ~ |w|^(-¾), an exact power of the radix.
   70|      8|    //  2. wʹ ← sw
   71|      8|    //  3. zʹ ← sz
   72|      8|    //  4. return zʹ * (wʹ.conjugate / wʹ.lengthSquared) (i.e. zʹ/wʹ).
   73|      8|    //
   74|      8|    //  Why is this safe and accurate? First, observe that wʹ and zʹ are both
   75|      8|    //  computed exactly because:
   76|      8|    //
   77|      8|    //  - s is an exact power of radix.
   78|      8|    //  - wʹ ~ |w|^(¼), and hence cannot overflow or underflow.
   79|      8|    //  - zʹ might overflow or underflow, but only if the final result also
   80|      8|    //       overflows or underflows. (This is more subtle than I make it
   81|      8|    //       sound. In particular, most of the fast ways one might try to
   82|      8|    //       compute s give rise to a situation where when |w| is close to
   83|      8|    //       one, multiplication by s is a dilation even though the actual
   84|      8|    //       division is a contraction or vice-versa, and thus intermediate
   85|      8|    //       computations might incorrectly overflow or underflow. Priest
   86|      8|    //       had to take some care to avoid this situation, but we do not,
   87|      8|    //       because we have already ruled out |w| ~ 1 before we call this
   88|      8|    //       function.)
   89|      8|    //
   90|      8|    //  Next observe that |wʹ.lengthSquared| ~ |w|^(½), so again this cannot
   91|      8|    //  overflow or underflow, and neither can (wʹ.conjugate/wʹ.lengthSquared),
   92|      8|    //  which has magnitude like |w|^(-¼).
   93|      8|    //
   94|      8|    //  Note that because the scale factor is always a power of the radix,
   95|      8|    //  the rescaling does not affect rounding, and so this algorithm is scale-
   96|      8|    //  invariant compared to the mainline `/` implementation, up to the
   97|      8|    //  underflow boundary.
   98|      8|    //
   99|      8|    //  Note that our final assembly of the result is different from Priest;
  100|      8|    //  he applies s to w twice, instead of once to w and once to z, and
  101|      8|    //  does the product as (zw̅ʺ)*(1/|wʹ|²), while we do zʹ(w̅ʹ/|wʹ|²). We
  102|      8|    //  prefer our version for three reasons:
  103|      8|    //
  104|      8|    //  1. it extracts a little more ILP
  105|      8|    //  2. it makes it so that we get exactly the same roundings on the
  106|      8|    //     rescaled divide path as on the fast path, so that z/w = tz/tw
  107|      8|    //     when tz and tw are computed exactly.
  108|      8|    //  3. it unlocks a future optimization where we hoist s and
  109|      8|    //     (w̅ʹ/|wʹ|²) and make divisions all fast-path without perturbing
  110|      8|    //     rounding.
  111|      8|    let s = RealType(
  112|      8|      sign: .plus,
  113|      8|      exponent: -3*w.magnitude.exponent/4,
  114|      8|      significand: 1
  115|      8|    )
  116|      8|    let wʹ = w.multiplied(by: s)
  117|      8|    let zʹ = z.multiplied(by: s)
  118|      8|    return zʹ * wʹ.conjugate.divided(by: wʹ.lengthSquared)
  119|     16|  }
