   12|       |extension FixedWidthInteger {
   13|       |  /// `~0` (all-ones) if this value is negative, otherwise `0`.
   14|       |  ///
   15|       |  /// Note that if `Self` is unsigned, this always returns `0`,
   16|       |  /// but it is useful for writing algorithms that are generic over
   17|       |  /// signed and unsigned integers.
   18|       |  @inline(__always) @usableFromInline
   19|     63|  var signbit: Self {
   20|     63|    return self < .zero ? ~.zero : .zero
   21|     63|  }
   22|       |  
   23|       |  /// Saturating integer addition
   24|       |  ///
   25|       |  /// `self + other` clamped to the representable range of the type. e.g.:
   26|       |  /// ```
   27|       |  /// let a: Int8 = 84
   28|       |  /// let b: Int8 = 100
   29|       |  /// // 84 + 100 = 184 is not representable as
   30|       |  /// // Int8, so `c` is clamped to Int8.max (127).
   31|       |  /// let c = a.addingWithSaturation(b)
   32|       |  /// ```
   33|       |  ///
   34|       |  /// If the "normal addition" `self + other` does not trap,
   35|       |  /// this method produces the same result.
   36|       |  @inlinable
   37|     19|  public func addingWithSaturation(_ other: Self) -> Self {
   38|     19|    let (wrapped, overflow) = addingReportingOverflow(other)
   39|     19|    return overflow ? Self.max &- signbit : wrapped
   40|     19|  }
   41|       |  
   42|       |  /// Saturating integer subtraction
   43|       |  ///
   44|       |  /// `self - other` clamped to the representable range of the type. e.g.:
   45|       |  /// ```
   46|       |  /// let a: UInt = 37
   47|       |  /// let b: UInt = 42
   48|       |  /// // 37 - 42 = -5, which is not representable as
   49|       |  /// // UInt, so `c` is clamped to UInt.min (zero).
   50|       |  /// let c = a.subtractingWithSaturation(b)
   51|       |  /// ```
   52|       |  ///
   53|       |  /// Note that `a.addingWithSaturation(-b)` is not always equivalent to
   54|       |  /// `a.subtractingWithSaturation(b)`, because `-b` is not representable
   55|       |  /// if `b` is the minimum value of a signed type.
   56|       |  ///
   57|       |  /// If the "normal subtraction" `self - other` does not trap,
   58|       |  /// this method produces the same result.
   59|       |  @inlinable
   60|     19|  public func subtractingWithSaturation(_ other: Self) -> Self {
   61|     19|    let (wrapped, overflow) = subtractingReportingOverflow(other)
   62|     19|    if !overflow { return wrapped }
   63|     10|    return Self.isSigned ? Self.max &- signbit : 0
   64|     19|  }
   65|       |  
   66|       |  /// Saturating integer negation
   67|       |  ///
   68|       |  /// For unsigned types, the result is always zero. This is not very
   69|       |  /// interesting, but may occasionally be useful in generic contexts.
   70|       |  /// For signed types, the result is `-self` unless `self` is `.min`,
   71|       |  /// in which case the result is `.max`.
   72|       |  @inlinable
   73|      9|  public func negatedWithSaturation() -> Self {
   74|      9|    Self.zero.subtractingWithSaturation(self)
   75|      9|  }
   76|       |  
   77|       |  /// Saturating integer multiplication
   78|       |  ///
   79|       |  /// `self * other` clamped to the representable range of the type. e.g.:
   80|       |  /// ```
   81|       |  /// let a: Int8 = -16
   82|       |  /// let b: Int8 = -8
   83|       |  /// // -16 * -8 = 128 is not representable as
   84|       |  /// // Int8, so `c` is clamped to Int8.max (127).
   85|       |  /// let c = a.multipliedWithSaturation(by: b)
   86|       |  /// ```
   87|       |  ///
   88|       |  /// If the "normal multiplication" `self * other` does not trap,
   89|       |  /// this method produces the same result.
   90|       |  @inlinable
   91|     18|  public func multipliedWithSaturation(by other: Self) -> Self {
   92|     18|    let (high, low) = multipliedFullWidth(by: other)
   93|     18|    let wrapped = Self(truncatingIfNeeded: low)
   94|     18|    if high == wrapped.signbit { return wrapped }
   95|     12|    return Self.max &- high.signbit
   96|     18|  }

   98|       |  /// Bitwise left shift with rounding and saturation.
   99|       |  ///
  100|       |  /// `self` multiplied by the rational number 2^(`count`), saturated to the
  101|       |  /// range `Self.min ... Self.max`, and rounded according to `rule`.
  102|       |  ///
  103|       |  /// See `shifted(rightBy:rounding:)`, defined on `BinaryInteger`, for more
  104|       |  /// discussion of rounding shifts with examples.
  105|       |  ///
  106|       |  /// - Parameters:
  107|       |  ///   - leftBy count: the number of bits to shift by. If positive, this is
  108|       |  ///     a left-shift, and if negative a right shift.
  109|       |  ///   - rounding rule: the direction in which to round if `count` is negative.
  110|       |  @inlinable
  111|       |  public func shiftedWithSaturation(
  112|       |    leftBy count: Int,
  113|       |    rounding rule: RoundingRule = .down
  114|     14|  ) -> Self {
  115|     14|    if count == 0 { return self }
  116|     12|    // If count is negative, negate it and do a right shift without
  117|     12|    // saturation instead, since we already have that implemented.
  118|     12|    guard count > 0 else {
  119|      2|      return shifted(
  120|      2|        rightBy: count.negatedWithSaturation(),
  121|      2|        rounding: rule
  122|      2|      )
  123|     10|    }
  124|     10|    let clamped = Self.max &- signbit
  125|     10|    guard count < Self.bitWidth else {
  126|      4|      // If count is bitWidth or greater, we always overflow
  127|      4|      // unless self is zero.
  128|      4|      return self == 0 ? 0 : clamped
  129|      6|    }
  130|      6|    // Now we have 0 < count < bitWidth, so we can use a nice
  131|      6|    // straightforward implementation; a shift overflows if
  132|      6|    // the complementary shift doesn't match sign-or-zero
  133|      6|    // extension. E.g.:
  134|      6|    //
  135|      6|    // - signed 0b0010_1111 << 2 overflows, because
  136|      6|    //   0b0010_1111 >> 5 is 0b0000_0001, which does not
  137|      6|    //   equal 0b0000_0000
  138|      6|    //
  139|      6|    // - unsigned 0b0010_1111 << 2 does not overflow,
  140|      6|    //   because 0b0010_0000 >> 6 is 0b0000_0000, which
  141|      6|    //   does equal 0b0000_0000.
  142|      6|    let valueBits = Self.bitWidth &- (Self.isSigned ? 1 : 0)
  143|      6|    let wrapped = self &<< count
  144|      6|    let complement = valueBits &- count
  145|      6|    return self &>> complement == signbit ? wrapped : clamped
  146|     14|  }
